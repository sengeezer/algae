## Problem

Given a linked list, block size `k`, and rotation count `d`, rotate each block of `k` nodes by `d` positions (left rotation), while leaving block boundaries intact.

## Approach

1. Split the list into blocks of size `k`.
2. For each block:
   - compute effective rotation `d % blockLength`
   - find new head after local rotation
   - reconnect rotated block to output
3. Continue for all blocks and stitch them together.

**Time Complexity**: O(n)
**Space Complexity**: O(1)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function rotateBlockWise(head, k, d) {
  if (head === null || k <= 1 || d === 0) return head;

  const dummy = new Node(0);
  let outputTail = dummy;
  let current = head;

  while (current !== null) {
    let blockHead = current;
    let blockTail = current;
    let length = 1;

    while (length < k && blockTail.next !== null) {
      blockTail = blockTail.next;
      length++;
    }

    const nextBlock = blockTail.next;
    blockTail.next = null;

    const shift = d % length;
    if (shift > 0) {
      let splitPrev = blockHead;
      for (let i = 1; i < shift; i++) {
        splitPrev = splitPrev.next;
      }

      const newHead = splitPrev.next;
      splitPrev.next = null;
      blockTail.next = blockHead;
      blockHead = newHead;
    }

    outputTail.next = blockHead;
    while (outputTail.next !== null) {
      outputTail = outputTail.next;
    }

    current = nextBlock;
  }

  return dummy.next;
}
```
