## Why It Surfaces

This problem is a good test of whether you recognize useful matrix monotonicity. Since each row is sorted, the top-right corner acts like a staircase: a `1` means the current row is at least as good as every row above it, while a `0` means the current row cannot beat any row that already reached farther left.

## Worked Example

For `[[0, 0, 1, 1], [0, 1, 1, 1], [0, 0, 0, 1]]`, start at the top-right `1`. That already makes row `0` the current best, so move left. When a `0` blocks further progress on that row, move down. Repeating that walk reaches row `1` and three trailing `1`s, so the answer is index `1`.

function validateSortedBinaryMatrix(matrix) {
  if (!Array.isArray(matrix) || matrix.length === 0 || !matrix.every((row) => Array.isArray(row))) {
    throw new TypeError('matrix must be a non-empty 2D array');
  }

  const width = matrix[0].length;
  if (width === 0) {
    throw new TypeError('matrix rows must not be empty');
  }

  for (const row of matrix) {
    if (row.length !== width || !row.every((value) => value === 0 || value === 1)) {
      throw new TypeError('matrix must contain rows of equal length with only 0 and 1 values');
    }

    for (let index = 1; index < row.length; index += 1) {
      if (row[index] < row[index - 1]) {
        throw new TypeError('each row must be sorted in nondecreasing order');
      }
    }
  }
}

function findTheRowWithMaximumNumberOf1s(matrix) {
  validateSortedBinaryMatrix(matrix);

  let bestRow = -1;
  let row = 0;
  let col = matrix[0].length - 1;

  while (row < matrix.length && col >= 0) {
    if (matrix[row][col] === 1) {
      bestRow = row;
      col -= 1;
    } else {
      row += 1;
    }
  }

  return bestRow;
}

console.assert(
  findTheRowWithMaximumNumberOf1s([
    [0, 0, 1, 1],
    [0, 1, 1, 1],
    [0, 0, 0, 1],
  ]) === 1,

);

console.assert(
  findTheRowWithMaximumNumberOf1s([
    [0, 0, 0],
    [0, 0, 0],
  ]) === -1,
  'matrix with no 1s should return -1',
);

## Common Pitfalls

- Counting every row fully even though the matrix order lets you do better.
- Forgetting that `-1` is the natural answer when no row contains a `1`.
- Applying the staircase walk to rows that are not actually sorted.
