## Problem

Given an unsorted singly linked list, remove duplicate values so only the first occurrence of each value remains.

## Approach

Use a `Set` of seen values. Traverse with `current` and `prev`. If `current.data` is already in the set, bypass `current`; otherwise add it to the set and advance `prev`.

**Time Complexity**: O(n) average
**Space Complexity**: O(n)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function removeDuplicatesFromUnsortedLinkedList(head) {
  if (head === null) {
    return null;
  }

  const seen = new Set();
  let current = head;
  let prev = null;

  while (current !== null) {
    if (seen.has(current.data)) {
      prev.next = current.next;
    } else {
      seen.add(current.data);
      prev = current;
    }

    current = current.next;
  }

  return head;
}
```

```typescript
class Node {
  data: number;
  next: Node | null;

  constructor(data: number) {
    this.data = data;
    this.next = null;
  }
}

function removeDuplicatesFromUnsortedLinkedList(head: Node | null): Node | null {
  if (head === null) {
    return null;
  }

  const seen = new Set<number>();
  let current: Node | null = head;
  let prev: Node | null = null;

  while (current !== null) {
    if (seen.has(current.data)) {
      prev!.next = current.next;
    } else {
      seen.add(current.data);
      prev = current;
    }

    current = current.next;
  }

  return head;
}
```
