## Problem

For each window size from `1` to `n`, compute the maximum among the minimum values of all subarrays of that size.

## Approach

Find the largest window where each element is minimum using previous/next smaller indices. Use that window length to update answers, then backfill missing sizes from the right.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function maxOfMins(arr) {
  const n = arr.length;
  const left = new Array(n);
  const right = new Array(n);
  const ans = new Array(n + 1).fill(-Infinity);
  const stack = [];

  for (let i = 0; i < n; i++) {
    while (stack.length && arr[stack[stack.length - 1]] >= arr[i]) stack.pop();
    left[i] = stack.length ? stack[stack.length - 1] : -1;
    stack.push(i);
  }

  stack.length = 0;

  for (let i = n - 1; i >= 0; i--) {
    while (stack.length && arr[stack[stack.length - 1]] >= arr[i]) stack.pop();
    right[i] = stack.length ? stack[stack.length - 1] : n;
    stack.push(i);
  }

  for (let i = 0; i < n; i++) {
    const len = right[i] - left[i] - 1;
    ans[len] = Math.max(ans[len], arr[i]);
  }

  for (let len = n - 1; len >= 1; len--) {
    ans[len] = Math.max(ans[len], ans[len + 1]);
  }

  return ans.slice(1);
}
```
