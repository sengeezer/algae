## Problem

Given a circular singly linked list, split it into two circular linked lists where sizes differ by at most one node.

## Approach

Use slow and fast pointers from head. Move fast by two nodes and slow by one node until fast reaches the end of the circle. Slow will be at midpoint. Rewire links:

- First half head: original `head`
- Second half head: `slow.next`
- End of first half points to first head
- End of second half points to second head

**Time Complexity**: O(n)
**Space Complexity**: O(1)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function splitCircularLinkedList(head) {
  if (head === null) {
    return [null, null];
  }

  if (head.next === head) {
    return [head, null];
  }

  let slow = head;
  let fast = head;

  while (fast.next !== head && fast.next.next !== head) {
    slow = slow.next;
    fast = fast.next.next;
  }

  if (fast.next.next === head) {
    fast = fast.next;
  }

  const head1 = head;
  const head2 = slow.next;

  slow.next = head1;
  fast.next = head2;

  return [head1, head2];
}
```

```typescript
class Node {
  data: number;
  next: Node | null;

  constructor(data: number) {
    this.data = data;
    this.next = null;
  }
}

function splitCircularLinkedList(head: Node | null): [Node | null, Node | null] {
  if (head === null) {
    return [null, null];
  }

  if (head.next === head) {
    return [head, null];
  }

  let slow: Node = head;
  let fast: Node = head;

  while (fast.next !== head && fast.next!.next !== head) {
    slow = slow.next!;
    fast = fast.next!.next!;
  }

  if (fast.next!.next === head) {
    fast = fast.next!;
  }

  const head1: Node = head;
  const head2: Node = slow.next!;

  slow.next = head1;
  fast.next = head2;

  return [head1, head2];
}
```
