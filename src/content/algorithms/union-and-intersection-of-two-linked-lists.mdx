## Problem

Given two linked lists, return:

- Union list: unique values present in either list
- Intersection list: unique values present in both lists

## Approach

Traverse the first list and record values in a set while building union candidates. Traverse the second list:

- Add unseen values to union
- Add values present in first-set to intersection (once)

Build result linked lists from these value sequences.

**Time Complexity**: O(m + n)
**Space Complexity**: O(m + n)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function buildList(values) {
  const dummy = new Node(0);
  let tail = dummy;

  for (const value of values) {
    tail.next = new Node(value);
    tail = tail.next;
  }

  return dummy.next;
}

function unionAndIntersection(head1, head2) {
  const seenUnion = new Set();
  const seenFirst = new Set();
  const seenIntersection = new Set();

  const unionValues = [];
  const intersectionValues = [];

  let current = head1;
  while (current !== null) {
    seenFirst.add(current.data);
    if (!seenUnion.has(current.data)) {
      seenUnion.add(current.data);
      unionValues.push(current.data);
    }
    current = current.next;
  }

  current = head2;
  while (current !== null) {
    if (!seenUnion.has(current.data)) {
      seenUnion.add(current.data);
      unionValues.push(current.data);
    }

    if (seenFirst.has(current.data) && !seenIntersection.has(current.data)) {
      seenIntersection.add(current.data);
      intersectionValues.push(current.data);
    }

    current = current.next;
  }

  return {
    unionHead: buildList(unionValues),
    intersectionHead: buildList(intersectionValues),
  };
}
```
