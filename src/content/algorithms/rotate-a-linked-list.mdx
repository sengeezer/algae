## Problem

Given a singly linked list and integer `k`, rotate the list counterclockwise by `k` nodes.

## Approach

Find list length and normalize `k = k % length`. If `k` is 0, return head. Otherwise, find the `k`-th node, break after it, and connect old tail to old head.

**Time Complexity**: O(n)
**Space Complexity**: O(1)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function rotateLinkedList(head, k) {
  if (head === null || head.next === null || k === 0) {
    return head;
  }

  let length = 1;
  let tail = head;

  while (tail.next !== null) {
    tail = tail.next;
    length++;
  }

  k = k % length;
  if (k === 0) {
    return head;
  }

  let current = head;
  for (let i = 1; i < k; i++) {
    current = current.next;
  }

  const newHead = current.next;
  current.next = null;
  tail.next = head;

  return newHead;
}
```

```typescript
class Node {
  data: number;
  next: Node | null;

  constructor(data: number) {
    this.data = data;
    this.next = null;
  }
}

function rotateLinkedList(head: Node | null, k: number): Node | null {
  if (head === null || head.next === null || k === 0) {
    return head;
  }

  let length = 1;
  let tail: Node = head;

  while (tail.next !== null) {
    tail = tail.next;
    length++;
  }

  k = k % length;
  if (k === 0) {
    return head;
  }

  let current: Node = head;
  for (let i = 1; i < k; i++) {
    current = current.next!;
  }

  const newHead = current.next;
  current.next = null;
  tail.next = head;

  return newHead;
}
```
