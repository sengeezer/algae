## Problem

Given a weighted directed graph, a source, a destination, and a required set of intermediate vertices, return the cheapest route that visits every required vertex at least once.

## Approach

Encode the visited required vertices as a bitmask and search state space with a queue ordered by accumulated cost. When the destination is reached with all required vertices visited, keep the minimum cost.

**Time Complexity**: O(V * 2^m)
**Space Complexity**: O(V * 2^m)

## Implementation

```javascript
function minimumCostPathWithRequiredNodes(graph, source, destination, requiredNodes) {
  const indexByNode = new Map(requiredNodes.map((node, index) => [node, index]));
  const targetMask = (1 << requiredNodes.length) - 1;
  const queue = [[source, 0, indexByNode.has(source) ? 1 << indexByNode.get(source) : 0]];
  const best = new Map();
  let head = 0;
  let answer = Number.POSITIVE_INFINITY;

  while (head < queue.length) {
    const [node, cost, mask] = queue[head++];
    const stateKey = `${node}:${mask}`;

    if (best.has(stateKey) && best.get(stateKey) <= cost) continue;
    best.set(stateKey, cost);

    if (node === destination && mask === targetMask) {
      answer = Math.min(answer, cost);
      continue;
    }

    for (const [nextNode, weight] of graph[node] ?? []) {
      const nextMask = indexByNode.has(nextNode)
        ? mask | (1 << indexByNode.get(nextNode))
        : mask;
      queue.push([nextNode, cost + weight, nextMask]);
    }
  }

  return Number.isFinite(answer) ? answer : -1;
}
```

```typescript
type WeightedEdge = [number, number];

type WeightedGraph = Array<WeightedEdge[]>;

function minimumCostPathWithRequiredNodes(
  graph: WeightedGraph,
  source: number,
  destination: number,
  requiredNodes: number[],
): number {
  const indexByNode = new Map(requiredNodes.map((node, index) => [node, index]));
  const targetMask = (1 << requiredNodes.length) - 1;
  const initialMask = indexByNode.has(source) ? 1 << indexByNode.get(source)! : 0;
  const queue: Array<[number, number, number]> = [[source, 0, initialMask]];
  const best = new Map<string, number>();
  let head = 0;
  let answer = Number.POSITIVE_INFINITY;

  while (head < queue.length) {
    const [node, cost, mask] = queue[head++];
    const stateKey = `${node}:${mask}`;

    if (best.has(stateKey) && best.get(stateKey)! <= cost) continue;
    best.set(stateKey, cost);

    if (node === destination && mask === targetMask) {
      answer = Math.min(answer, cost);
      continue;
    }

    for (const [nextNode, weight] of graph[node] ?? []) {
      const nextMask = indexByNode.has(nextNode)
        ? mask | (1 << indexByNode.get(nextNode)!)
        : mask;
      queue.push([nextNode, cost + weight, nextMask]);
    }
  }

  return Number.isFinite(answer) ? answer : -1;
}
```
