## Problem

For each element, find nearest greater index on left and right. Return the maximum product of those indices.

## Approach

Use monotonic stacks to compute left and right nearest-greater indices, then scan for max product.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function maxIndexProduct(arr) {
  const n = arr.length;
  const left = new Array(n).fill(0);
  const right = new Array(n).fill(0);
  const stack = [];

  for (let i = 0; i < n; i++) {
    while (stack.length && arr[stack[stack.length - 1]] <= arr[i]) stack.pop();
    left[i] = stack.length ? stack[stack.length - 1] + 1 : 0;
    stack.push(i);
  }

  stack.length = 0;

  for (let i = n - 1; i >= 0; i--) {
    while (stack.length && arr[stack[stack.length - 1]] <= arr[i]) stack.pop();
    right[i] = stack.length ? stack[stack.length - 1] + 1 : 0;
    stack.push(i);
  }

  let best = 0;
  for (let i = 0; i < n; i++) {
    best = Math.max(best, left[i] * right[i]);
  }

  return best;
}
```

```typescript
  stack.length = 0;

  for (let i = n - 1; i >= 0; i--) {
    while (stack.length > 0 && arr[stack[stack.length - 1]] <= arr[i]) stack.pop();
    right[i] = stack.length > 0 ? stack[stack.length - 1] + 1 : 0;
    stack.push(i);
  }

  let best = 0;
  for (let i = 0; i < n; i++) {
    best = Math.max(best, left[i] * right[i]);
  }

  return best;
}
```
