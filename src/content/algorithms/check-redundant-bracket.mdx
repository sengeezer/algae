## Problem

Given an arithmetic expression, determine whether it contains redundant brackets.

## Approach

Push characters to a stack. On `)`, pop until `(` and check whether at least one operator was inside.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function hasRedundantBracket(expr) {
  const stack = [];
  const operators = new Set(['+', '-', '*', '/']);

  for (const ch of expr) {
    if (ch === ')') {
      let hasOperator = false;

      while (stack.length && stack[stack.length - 1] !== '(') {
        const top = stack.pop();
        if (operators.has(top)) hasOperator = true;
      }

      stack.pop();
      if (!hasOperator) return true;
    } else {
      stack.push(ch);
    }
  }

  return false;
}
```

```typescript
function hasRedundantBracket(expr: string): boolean {
  const stack: string[] = [];
  const operators = new Set(['+', '-', '*', '/']);

  for (const ch of expr) {
    if (ch === ')') {
      let hasOperator = false;

      while (stack.length > 0 && stack[stack.length - 1] !== '(') {
        const top = stack.pop()!;
        if (operators.has(top)) hasOperator = true;
      }

      stack.pop();
      if (!hasOperator) return true;
    } else {
      stack.push(ch);
    }
  }

  return false;
}
```
