## Problem

Sort a doubly linked list in ascending order.

## Approach

Use merge sort:

1. Find midpoint with slow/fast pointers.
2. Split into left and right halves.
3. Recursively sort each half.
4. Merge sorted halves while setting both `next` and `prev` pointers.

**Time Complexity**: O(n log n)
**Space Complexity**: O(log n)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.prev = null;
    this.next = null;
  }
}

function split(head) {
  let slow = head;
  let fast = head;

  while (fast.next !== null && fast.next.next !== null) {
    slow = slow.next;
    fast = fast.next.next;
  }

  const second = slow.next;
  slow.next = null;
  if (second !== null) {
    second.prev = null;
  }

  return second;
}

function merge(left, right) {
  if (left === null) return right;
  if (right === null) return left;

  if (left.data <= right.data) {
    left.next = merge(left.next, right);
    if (left.next !== null) {
      left.next.prev = left;
    }
    left.prev = null;
    return left;
  }

  right.next = merge(left, right.next);
  if (right.next !== null) {
    right.next.prev = right;
  }
  right.prev = null;
  return right;
}

function mergeSortDoublyLinkedList(head) {
  if (head === null || head.next === null) {
    return head;
  }

  const second = split(head);

  const left = mergeSortDoublyLinkedList(head);
  const right = mergeSortDoublyLinkedList(second);

  return merge(left, right);
}
```

class Node {
  data: number;
  prev: Node | null;
  next: Node | null;

  constructor(data: number) {
    this.data = data;
    this.prev = null;
    this.next = null;
  }
}

function split(head: Node): Node | null {
  let slow: Node = head;
  let fast: Node = head;

  while (fast.next !== null && fast.next.next !== null) {
    slow = slow.next!;
    fast = fast.next.next;
  }

  const second = slow.next;
  slow.next = null;
  if (second !== null) {
    second.prev = null;
  }

  return second;
}

function merge(left: Node | null, right: Node | null): Node | null {
  if (left === null) return right;
  if (right === null) return left;

  if (left.data <= right.data) {
    left.next = merge(left.next, right);
    if (left.next !== null) {
      left.next.prev = left;
    }
    left.prev = null;
    return left;
  }

  right.next = merge(left, right.next);
  if (right.next !== null) {
    right.next.prev = right;
  }
  right.prev = null;
  return right;
}

function mergeSortDoublyLinkedList(head: Node | null): Node | null {
  if (head === null || head.next === null) {
    return head;
  }

  const second = split(head);

  const left = mergeSortDoublyLinkedList(head);
  const right = mergeSortDoublyLinkedList(second);

  return merge(left, right);
}
