## Problem

Convert an infix expression to postfix expression.

## Approach

Scan characters left to right:
- append operands directly,
- push `(`,
- on `)` pop until `(`,
- for operators, pop while stack top has higher or equal precedence.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function precedence(op) {
  if (op === '^') return 3;
  if (op === '*' || op === '/') return 2;
  if (op === '+' || op === '-') return 1;
  return 0;
}

function infixToPostfix(expr) {
  const stack = [];
  let out = '';

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      out += ch;
    } else if (ch === '(') {
      stack.push(ch);
    } else if (ch === ')') {
      while (stack.length && stack[stack.length - 1] !== '(') {
        out += stack.pop();
      }
      stack.pop();
    } else {
      while (
        stack.length &&
        precedence(stack[stack.length - 1]) >= precedence(ch)
      ) {
        out += stack.pop();
      }
      stack.push(ch);
    }
  }

  while (stack.length) out += stack.pop();
  return out;
}
```

function precedence(op: string): number {
  if (op === '^') return 3;
  if (op === '*' || op === '/') return 2;
  if (op === '+' || op === '-') return 1;
  return 0;
}

function infixToPostfix(expr: string): string {
  const stack: string[] = [];
  let out = '';

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      out += ch;
    } else if (ch === '(') {
      stack.push(ch);
    } else if (ch === ')') {
      while (stack.length && stack[stack.length - 1] !== '(') {
        out += stack.pop();
      }
      stack.pop();
    } else {
      while (stack.length && precedence(stack[stack.length - 1]) >= precedence(ch)) {
        out += stack.pop()!;
      }
      stack.push(ch);
    }
  }

  while (stack.length) out += stack.pop()!;
  return out;
}
