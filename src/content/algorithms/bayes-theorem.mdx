## Why It Surfaces

Bayes' theorem is the standard way to reverse a conditional probability when you know a prior and a likelihood. In interview-style probability questions, the algorithmic part is usually computing the denominator correctly by summing all weighted explanations for the observed evidence.

## Worked Example

Assume a disease has prevalence `0.01`, a test has true-positive rate `0.95`, and the false-positive rate is `0.05`. The posterior probability of having the disease after a positive result is `0.95 * 0.01 / (0.95 * 0.01 + 0.05 * 0.99)`, which is about `0.161`.

function bayesTheorem(prior, likelihood, evidenceProbability) {
  for (const value of [prior, likelihood, evidenceProbability]) {
    if (typeof value !== 'number' || !Number.isFinite(value)) {
      throw new TypeError('All inputs must be finite numbers');
    }
  }

  if (prior < 0 || prior > 1 || likelihood < 0 || likelihood > 1) {
    throw new RangeError('prior and likelihood must be between 0 and 1');
  }

  if (evidenceProbability <= 0 || evidenceProbability > 1) {
    throw new RangeError('evidenceProbability must be between 0 and 1');
  }

  const numerator = prior * likelihood;

  if (numerator > evidenceProbability) {
    throw new RangeError('prior * likelihood cannot exceed evidenceProbability');
  }

  return numerator / evidenceProbability;
}

function posteriorFromPartition(priors, likelihoods, targetIndex) {
  if (!Array.isArray(priors) || !Array.isArray(likelihoods)) {
    throw new TypeError('priors and likelihoods must be arrays');
  }

  if (priors.length === 0 || priors.length !== likelihoods.length) {
    throw new RangeError('priors and likelihoods must be non-empty arrays of equal length');
  }

  if (!Number.isInteger(targetIndex) || targetIndex < 0 || targetIndex >= priors.length) {
    throw new RangeError('targetIndex must point to a valid hypothesis');
  }

  let priorSum = 0;
  let evidenceProbability = 0;

  for (let index = 0; index < priors.length; index += 1) {
    const prior = priors[index];
    const likelihood = likelihoods[index];

    if (typeof prior !== 'number' || typeof likelihood !== 'number') {
      throw new TypeError('All priors and likelihoods must be numbers');
    }

    if (!Number.isFinite(prior) || !Number.isFinite(likelihood)) {
      throw new TypeError('All priors and likelihoods must be finite numbers');
    }

    if (prior < 0 || prior > 1 || likelihood < 0 || likelihood > 1) {
      throw new RangeError('All priors and likelihoods must be between 0 and 1');
    }

    priorSum += prior;
    evidenceProbability += prior * likelihood;
  }

  if (Math.abs(priorSum - 1) > 1e-12) {
    throw new RangeError('Priors must sum to 1');
  }

  return bayesTheorem(priors[targetIndex], likelihoods[targetIndex], evidenceProbability);
}

function approximatelyEqual(left, right, epsilon = 1e-12) {
  return Math.abs(left - right) <= epsilon;
}

  'posterior should match weighted partition update'
);
console.assert(
  approximatelyEqual(bayesTheorem(1 / 6, 3 / 4, 1 / 3), 3 / 8),
  'liar die example should produce 3/8'

## Common Pitfalls

- Plugging `P(B)` into the numerator instead of `P(B given A)`.
- Forgetting to compute the evidence probability from all competing hypotheses.
- Treating Bayes' theorem as necessary when direct conditional probability already answers the question.
