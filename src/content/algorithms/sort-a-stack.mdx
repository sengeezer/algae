## Problem

Sort a stack using recursion and stack operations only.

## Approach

Pop one element, sort remaining stack recursively, then insert popped element in sorted order.

**Time Complexity**: O(n^2)
**Space Complexity**: O(n)

## Implementation

```javascript
function sortedInsert(stack, value) {
  if (stack.length === 0 || stack[stack.length - 1] <= value) {
    stack.push(value);
    return;
  }

  const top = stack.pop();
  sortedInsert(stack, value);
  stack.push(top);
}

function sortStack(stack) {
  if (stack.length === 0) return;

  const top = stack.pop();
  sortStack(stack);
  sortedInsert(stack, top);
}
```

```typescript

  if (stack.length === 0 || stack[stack.length - 1] <= value) {
    stack.push(value);
    return;
  }

  const top = stack.pop()!;

  stack.push(top);

function sortStack(stack: number[]): void {
  if (stack.length === 0) return;

  const top = stack.pop()!;
  sortStack(stack);
  sortedInsert(stack, top);
}
```
