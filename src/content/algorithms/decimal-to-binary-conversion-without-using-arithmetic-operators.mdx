## Overview

Convert a non-negative decimal integer `n` to binary without using arithmetic operators like `%` or `/` for digit extraction.

Examples:

- `10` -> `1010`
- `38` -> `100110`

Use:

- `(n & 1)` to read the least significant bit
- `n >> 1` to drop the least significant bit

## Algorithm

1. If `n` is `0`, return `"0"`.
2. Repeatedly inspect `(n & 1)` and prepend `0` or `1`.
3. Right shift `n` by one (`n >>= 1`).
4. Continue until `n` becomes `0`.

```js
function decimalToBinaryWithoutArithmetic(n) {
  if (!Number.isInteger(n) || n < 0) {
    throw new Error("Input must be a non-negative integer");
  }

  if (n === 0) return "0";

  let value = n;
  let bin = "";

  while (value > 0) {
    bin = ((value & 1) === 1 ? "1" : "0") + bin;
    value >>= 1;
  }

  return bin;
}

// Optional variant using built-in conversion for comparison
function decimalToBinaryBuiltIn(n) {
  if (!Number.isInteger(n) || n < 0) {
    throw new Error("Input must be a non-negative integer");
  }
  return n.toString(2);
}

console.assert(decimalToBinaryWithoutArithmetic(10) === "1010", "10 -> 1010");
console.assert(decimalToBinaryWithoutArithmetic(38) === "100110", "38 -> 100110");
console.assert(decimalToBinaryWithoutArithmetic(0) === "0", "0 -> 0");
console.assert(
  decimalToBinaryWithoutArithmetic(255) === decimalToBinaryBuiltIn(255),
  "bitwise conversion should match built-in"
);
```

## Notes

- This approach mirrors manual base-2 conversion but uses bit-level primitives.
- In JavaScript, bitwise operators work on 32-bit signed integers; for larger values, use `BigInt` and `& 1n`, `>> 1n`.
