## Why It Surfaces

The direct inverse-product CRT formula is compact, but many implementations and proofs work by combining equations two at a time. That pairwise view is useful when you want to understand how the final remainder is assembled incrementally.

## Worked Example

Combine `x % 3 = 2` with `x % 5 = 3`. The smallest positive value satisfying both is `8`, so the two equations collapse into `x % 15 = 8`. Repeating that same merge process over a longer list eventually yields one final congruence modulo the full product.

function normalizeModulo(value, modulus) {
  return ((value % modulus) + modulus) % modulus;
}

function gcdExtended(a, b) {
  if (a === 0) {
    return { gcd: b, x: 0, y: 1 };
  }

  const { gcd, x, y } = gcdExtended(b % a, a);
  return {
    gcd,
    x: y - Math.floor(b / a) * x,
    y: x,
  };
}

function validateCongruenceSystem(remainders, moduli) {
  if (!Array.isArray(remainders) || !Array.isArray(moduli) || remainders.length !== moduli.length || remainders.length === 0) {
    throw new TypeError('remainders and moduli must be non-empty arrays of equal length');
  }

  if (!remainders.every((value) => Number.isInteger(value))) {
    throw new RangeError('remainders must be integers');
  }

  if (!moduli.every((value) => Number.isInteger(value) && value > 1)) {
    throw new RangeError('moduli must be integers greater than 1');
  }
}

function combineTwoCongruences(leftRemainder, leftModulus, rightRemainder, rightModulus) {
  const { gcd, x } = gcdExtended(leftModulus, rightModulus);

  if (gcd !== 1) {
    throw new RangeError('moduli must remain pairwise coprime during combination');
  }

  const delta = normalizeModulo(rightRemainder - leftRemainder, rightModulus);
  const multiplier = normalizeModulo(delta * x, rightModulus);
  const modulus = leftModulus * rightModulus;
  const remainder = normalizeModulo(leftRemainder + leftModulus * multiplier, modulus);

  return { remainder, modulus };
}

function combineModularEquations(remainders, moduli) {
  validateCongruenceSystem(remainders, moduli);

  let currentRemainder = normalizeModulo(remainders[0], moduli[0]);
  let currentModulus = moduli[0];

  for (let index = 1; index < moduli.length; index += 1) {
    const combined = combineTwoCongruences(
      currentRemainder,
      currentModulus,
      normalizeModulo(remainders[index], moduli[index]),
      moduli[index],
    );

    currentRemainder = combined.remainder;
    currentModulus = combined.modulus;
  }

  return currentRemainder === 0 ? currentModulus : currentRemainder;
}

function combineTwoCongruences(
  leftRemainder: number,
  leftModulus: number,
  rightRemainder: number,
  rightModulus: number,
): { remainder: number; modulus: number } {
  const { gcd, x } = gcdExtended(leftModulus, rightModulus);

  if (gcd !== 1) {
    throw new RangeError('moduli must remain pairwise coprime during combination');
  }

  const delta = normalizeModulo(rightRemainder - leftRemainder, rightModulus);
  const multiplier = normalizeModulo(delta * x, rightModulus);
  const modulus = leftModulus * rightModulus;
  const remainder = normalizeModulo(leftRemainder + leftModulus * multiplier, modulus);

  return { remainder, modulus };
}

export function combineModularEquations(remainders: number[], moduli: number[]): number {
  validateCongruenceSystem(remainders, moduli);

  let currentRemainder = normalizeModulo(remainders[0], moduli[0]);
  let currentModulus = moduli[0];

  for (let index = 1; index < moduli.length; index += 1) {
    const combined = combineTwoCongruences(
      currentRemainder,
      currentModulus,
      normalizeModulo(remainders[index], moduli[index]),
      moduli[index],
    );

    currentRemainder = combined.remainder;
    currentModulus = combined.modulus;
  }

  return currentRemainder === 0 ? currentModulus : currentRemainder;
}

## Follow-up Questions

- Return the final combined congruence as both `remainder` and `modulus` instead of only the smallest positive representative.
- Extend the merge step to handle some non-coprime congruence systems that still have solutions.
- Compare this pairwise combiner with the full-product inverse-based CRT formula from the related implementation.

## Common Pitfalls

- Forgetting that each merge step still requires coprime moduli in this simplified form.
- Returning intermediate remainder `0` as the modulus too early and breaking later combinations.
- Mixing the order of the remainders and moduli arrays when folding the equations.
