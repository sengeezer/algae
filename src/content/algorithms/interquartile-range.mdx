## Problem

Calculate the interquartile range (IQR) of a dataset, which represents the range containing the middle 50% of values.

### Examples

**Example 1:**
```
Input: [1, 2, 3, 4, 5, 6, 7, 8, 9]

Q1 (25th percentile) = 3
Q3 (75th percentile) = 7
IQR = Q3 - Q1 = 4
```

**Example 2:**
```
Input: [10, 15, 20, 25, 30, 35, 40, 45, 50]

Q1 = 20
Q3 = 40
IQR = 20
```

## Approach

### Quartile Calculation:
1. Sort the data
2. Find Q1: value at 25th percentile
3. Find Q3: value at 75th percentile
4. IQR = Q3 - Q1

### Different Methods:
- **Inclusive:** Include median in both halves
- **Exclusive:** Exclude median from both halves
- **Linear Interpolation:** Interpolate between values

## Solution

### JavaScript

```javascript
/**
 * Calculate IQR with linear interpolation
 * Time: O(n log n), Space: O(n)
 */
function calculateIQR(data) {
  const sorted = [...data].sort((a, b) => a - b);
  const n = sorted.length;

  // Calculate Q1 (25th percentile)
  const q1Position = (n + 1) * 0.25 - 1; // 0-indexed
  const q1 = interpolatePercentile(sorted, q1Position);

  // Calculate Q3 (75th percentile)
  const q3Position = (n + 1) * 0.75 - 1; // 0-indexed
  const q3 = interpolatePercentile(sorted, q3Position);

  return {
    q1,
    q3,
    iqr: q3 - q1,
  };
}

/**
 * Linear interpolation for percentile value
 */
function interpolatePercentile(sorted, position) {
  const lower = Math.floor(position);
  const upper = Math.ceil(position);

  if (lower === upper) {
    return sorted[lower];
  }

  const weight = position - lower;
  return sorted[lower] * (1 - weight) + sorted[upper] * weight;
}

/**
 * Exclusive method (exclude median)
 * Time: O(n log n), Space: O(n)
 */
function calculateIQRExclusive(data) {
  const sorted = [...data].sort((a, b) => a - b);
  const n = sorted.length;

  // Split data into lower and upper halves
  const midIndex = Math.floor(n / 2);
  const lowerHalf = sorted.slice(0, midIndex);
  const upperHalf = sorted.slice(n % 2 === 0 ? midIndex : midIndex + 1);

  const q1 = getMedian(lowerHalf);
  const q3 = getMedian(upperHalf);

  return {
    q1,
    q3,
    iqr: q3 - q1,
  };
}

/**
 * Inclusive method (include median)
 * Time: O(n log n), Space: O(n)
 */
function calculateIQRInclusive(data) {
  const sorted = [...data].sort((a, b) => a - b);
  const n = sorted.length;

  // Include median in both halves
  const midIndex = Math.ceil(n / 2);
  const lowerHalf = sorted.slice(0, midIndex);
  const upperHalf = sorted.slice(Math.floor(n / 2));

  const q1 = getMedian(lowerHalf);
  const q3 = getMedian(upperHalf);

  return {
    q1,
    q3,
    iqr: q3 - q1,
  };
}

/**
 * Calculate median of array
 */
function getMedian(sorted) {
  const n = sorted.length;
  if (n % 2 === 1) {
    return sorted[Math.floor(n / 2)];
  }
  return (sorted[n / 2 - 1] + sorted[n / 2]) / 2;
}

/**
 * Identify outliers using IQR method
 * Outliers are values < Q1 - 1.5*IQR or > Q3 + 1.5*IQR
 */
function findOutliers(data) {
  const { q1, q3, iqr } = calculateIQR(data);
  const lowerBound = q1 - 1.5 * iqr;
  const upperBound = q3 + 1.5 * iqr;

  const outliers = data.filter((x) => x < lowerBound || x > upperBound);

  return {
    outliers,
    lowerBound,
    upperBound,
    count: outliers.length,
  };
}

/**
 * Get summary statistics including IQR
 */
function getQuartilesAndStats(data) {
  const sorted = [...data].sort((a, b) => a - b);
  const n = sorted.length;
  const min = sorted[0];
  const max = sorted[n - 1];
  const median = getMedian(sorted);

  const { q1, q3, iqr } = calculateIQR(data);

  return {
    min,
    q1,
    median,
    q3,
    max,
    iqr,
    range: max - min,
  };
}
```

### TypeScript

```typescript
interface QuartileResult {
  q1: number;
  q3: number;
  iqr: number;
}

interface OutlierResult {
  outliers: number[];
  lowerBound: number;
  upperBound: number;
  count: number;
}

interface FullStats extends QuartileResult {
  min: number;
  max: number;
  median: number;
  range: number;
}

/**
 * Calculate IQR with linear interpolation
 * Time: O(n log n), Space: O(n)
 */
function calculateIQR(data: number[]): QuartileResult {
  const sorted = [...data].sort((a, b) => a - b);
  const n = sorted.length;

  const q1Position = (n + 1) * 0.25 - 1;
  const q1 = interpolatePercentile(sorted, q1Position);

  const q3Position = (n + 1) * 0.75 - 1;
  const q3 = interpolatePercentile(sorted, q3Position);

  return {
    q1,
    q3,
    iqr: q3 - q1,
  };
}

/**
 * Linear interpolation for percentile
 */
function interpolatePercentile(sorted: number[], position: number): number {
  const lower = Math.floor(position);
  const upper = Math.ceil(position);

  if (lower === upper) {
    return sorted[lower];
  }

  const weight = position - lower;
  return sorted[lower] * (1 - weight) + sorted[upper] * weight;
}

/**
 * Find outliers using IQR method
 */
function findOutliers(data: number[]): OutlierResult {
  const { q1, q3, iqr } = calculateIQR(data);
  const lowerBound = q1 - 1.5 * iqr;
  const upperBound = q3 + 1.5 * iqr;

  const outliers = data.filter((x) => x < lowerBound || x > upperBound);

  return {
    outliers,
    lowerBound,
    upperBound,
    count: outliers.length,
  };
}

/**
 * Get complete statistics
 */
function getQuartilesAndStats(data: number[]): FullStats {
  const sorted = [...data].sort((a, b) => a - b);
  const n = sorted.length;

  const median = n % 2 === 1 ? sorted[n >> 1] : (sorted[n / 2 - 1] + sorted[n / 2]) / 2;
  const { q1, q3, iqr } = calculateIQR(data);

  return {
    min: sorted[0],
    q1,
    median,
    q3,
    max: sorted[n - 1],
    iqr,
    range: sorted[n - 1] - sorted[0],
  };
}
```

## Key Properties

- **Robust to Outliers:** Unlike range or standard deviation
- **50% Coverage:** Contains middle half of data
- **Outlier Detection:** Standard rule uses 1.5 × IQR boundaries
- **Distribution-Free:** No assumption about data distribution

## Complexity Analysis

- **Time:** O(n log n) dominated by sorting
- **Space:** O(n) for sorted copy of data
- **Calculation:** O(1) after sorting
