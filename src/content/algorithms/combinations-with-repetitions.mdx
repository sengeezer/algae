## Why It Surfaces

This is the clean multiset-selection pattern. The difference from ordinary combinations is small but decisive: after choosing a value, the next recursive call stays on the same index so that value can be used again.

## Worked Example

Choosing 2 scoops from `['vanilla', 'chocolate', 'strawberry']` allows repeats, so `['vanilla', 'vanilla']` is valid and order still does not matter. The count is `C(3 + 2 - 1, 2) = C(4, 2) = 6`, matching the six generated results.

function binomialCoefficient(n, k) {
  if (k < 0 || k > n) {
    return 0;
  }

  const choose = Math.min(k, n - k);
  let result = 1;

  for (let step = 1; step <= choose; step += 1) {
    result = (result * (n - choose + step)) / step;
  }

  return result;
}

  if (!Number.isInteger(n) || !Number.isInteger(r) || n < 0 || r < 0) {

  if (n === 0) {
    return r === 0 ? 1 : 0;

function combinationsWithRepetition(values, choose) {
  if (!Array.isArray(values) || !Number.isInteger(choose) || choose < 0) {
    return [];
  }

  if (choose === 0) {
    return [[]];
  }

  const combinations = [];
  const path = [];

  function backtrack(start) {
    if (path.length === choose) {
      combinations.push([...path]);
      return;
    }

    for (let index = start; index < values.length; index += 1) {
      path.push(values[index]);
      backtrack(index);
      path.pop();
    }
  }

  backtrack(0);
  return combinations;
}

export function combinationsWithRepetition<T>(values: readonly T[], choose: number): T[][] {
  if (!Number.isInteger(choose) || choose < 0) {
    return [];
  }

  if (choose === 0) {
    return [[]];
  }

  const combinations: T[][] = [];
  const path: T[] = [];

  function backtrack(start: number): void {
    if (path.length === choose) {
      combinations.push([...path]);
      return;
    }

    for (let index = start; index < values.length; index += 1) {
      path.push(values[index]);
      backtrack(index);
      path.pop();
    }
  }

  backtrack(0);
  return combinations;
}

## Follow-up Questions

- Prove the count formula with a stars-and-bars argument.
- Compare this generator with the no-repetition version that recurses into `index + 1`.
- Stream the combinations lazily instead of storing them all at once.

## Common Pitfalls

- Advancing to `index + 1` after taking a value and accidentally forbidding repeats.
- Treating different orders of the same multiset as separate answers.
- Forgetting the `n + r - 1` shift in the counting formula.
