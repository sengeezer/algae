## Problem

Implement `k` separate queues inside one backing array while supporting enqueue, dequeue, and emptiness checks.

## Approach

Use one array for values, one `next` array to link either queue nodes or free slots, plus `front`, `rear`, and `freeIndex` bookkeeping.

**Time Complexity**: O(1) enqueue/dequeue
**Space Complexity**: O(n + k)

## Implementation

```javascript
class KQueues {
  constructor(size, queueCount) {
    this.size = size;
    this.queueCount = queueCount;
    this.values = new Array(size);
    this.front = new Array(queueCount).fill(-1);
    this.rear = new Array(queueCount).fill(-1);
    this.next = new Array(size);

    for (let i = 0; i < size - 1; i += 1) {
      this.next[i] = i + 1;
    }
    this.next[size - 1] = -1;
    this.freeIndex = 0;
  }

  enqueue(queueIndex, value) {
    if (this.freeIndex === -1) return false;

    const slot = this.freeIndex;
    this.freeIndex = this.next[slot];
    this.values[slot] = value;
    this.next[slot] = -1;

    if (this.front[queueIndex] === -1) {
      this.front[queueIndex] = slot;
    } else {
      this.next[this.rear[queueIndex]] = slot;
    }

    this.rear[queueIndex] = slot;
    return true;
  }

  dequeue(queueIndex) {
    const frontIndex = this.front[queueIndex];
    if (frontIndex === -1) return null;

    this.front[queueIndex] = this.next[frontIndex];
    if (this.front[queueIndex] === -1) this.rear[queueIndex] = -1;

    this.next[frontIndex] = this.freeIndex;
    this.freeIndex = frontIndex;
    return this.values[frontIndex];
  }
}
```
