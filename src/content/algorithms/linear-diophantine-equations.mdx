## Why It Surfaces

This is where gcd becomes more than a divisor check. Once you know that `gcd(a, b)` divides every value of `ax + by`, the whole feasibility question collapses to one divisibility test, and extended Euclid gives the witness pair immediately.

## Worked Example

For `3x + 6y = 9`, `gcd(3, 6) = 3`, and `3` divides `9`, so a solution exists. One valid pair is `x = 3`, `y = 0`.

For `3x + 6y = 8`, the same gcd is `3`, but `3` does not divide `8`, so no integer solution can exist.

function extendedGcd(a, b) {
  let oldR = Math.abs(a);
  let r = Math.abs(b);
  let oldS = 1;
  let s = 0;
  let oldT = 0;
  let t = 1;

  while (r !== 0) {
    const quotient = Math.floor(oldR / r);
    [oldR, r] = [r, oldR - quotient * r];
    [oldS, s] = [s, oldS - quotient * s];
    [oldT, t] = [t, oldT - quotient * t];
  }

  return {
    gcd: oldR,
    x: a < 0 ? -oldS : oldS,
    y: b < 0 ? -oldT : oldT,
  };
}

function linearDiophantineEquations(a, b, c) {
  if (![a, b, c].every((value) => Number.isInteger(value))) {
    throw new TypeError('Coefficients and target must be integers');
  }

  if (a === 0 && b === 0) {
    throw new RangeError('At least one coefficient must be non-zero');
  }

  const { gcd, x, y } = extendedGcd(a, b);

  if (c % gcd !== 0) {
    return {
      hasSolution: false,
      gcd,
      x: null,
      y: null,
      stepX: null,
      stepY: null,
    };
  }

  const scale = c / gcd;

  return {
    hasSolution: true,
    gcd,
    x: x * scale,
    y: y * scale,
    stepX: b / gcd,
    stepY: -a / gcd,
  };
}

console.assert(first.hasSolution === true, '3x + 6y = 9 should be solvable');
console.assert(first.x !== null && first.y !== null && 3 * first.x + 6 * first.y === 9, 'Returned pair should satisfy the equation');

const second = linearDiophantineEquations(3, 6, 8);
console.assert(second.hasSolution === false, '3x + 6y = 8 should be unsolvable');

const third = linearDiophantineEquations(2, 5, 1);
console.assert(third.hasSolution === true, '2x + 5y = 1 should be solvable');

## Common Pitfalls

- Brute-forcing candidate `x` and `y` pairs instead of using gcd divisibility.
- Forgetting that the input coefficients may be negative, so the returned witness must respect the original signs.
- Returning only a boolean and omitting the particular solution even though extended Euclid already computed it.
