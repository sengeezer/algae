## Why It Surfaces

Primitive-root questions show up when the interviewer wants a concrete generator of the non-zero residues modulo a prime. The key idea is that a candidate `g` is a primitive root modulo prime `p` iff for every distinct prime factor `q` of `p - 1`, the value `g^((p - 1) / q) mod p` is not `1`.

## Worked Example

For `p = 7`, we have `phi(7) = 6`, whose distinct prime factors are `2` and `3`. Candidate `2` fails because `2^(6 / 2) mod 7 = 1`, but candidate `3` passes both checks, so the smallest primitive root is `3`.

function isPrime(value) {
  if (value <= 1) {
    return false;
  }

  if (value <= 3) {
    return true;
  }

  if (value % 2 === 0 || value % 3 === 0) {
    return false;
  }

    if (value % divisor === 0 || value % (divisor + 2) === 0) {
      return false;
    }
  }

  return true;
}

function modPow(base, exponent, mod) {
  let value = ((base % mod) + mod) % mod;
  let power = exponent;
  let result = 1;

  while (power > 0) {
    if (power & 1) {
      result = (result * value) % mod;
    }

    value = (value * value) % mod;
    power >>= 1;
  }

  return result;
}

function distinctPrimeFactors(value) {
  const factors = [];
  let remaining = value;

  for (let divisor = 2; divisor * divisor <= remaining; divisor += 1) {
    if (remaining % divisor !== 0) {
      continue;
    }

    factors.push(divisor);
    while (remaining % divisor === 0) {
      remaining = Math.floor(remaining / divisor);
    }
  }

  if (remaining > 1) {
    factors.push(remaining);
  }

  return factors;
}

function primitiveRootOfPrimeNumberModulo(prime) {
  if (!Number.isInteger(prime) || !isPrime(prime)) {
    return -1;
  }

  if (prime === 2) {
    return 1;
  }

  const phi = prime - 1;
  const factors = distinctPrimeFactors(phi);

  for (let candidate = 2; candidate < prime; candidate += 1) {
    let isPrimitiveRoot = true;

    for (const factor of factors) {
      if (modPow(candidate, Math.floor(phi / factor), prime) === 1) {
        isPrimitiveRoot = false;
        break;
      }
    }

    if (isPrimitiveRoot) {
      return candidate;
    }
  }

  return -1;
}

## Follow-up Questions

- Return all primitive roots instead of only the smallest one.
- Explain why only the distinct prime factors of `p - 1` are needed in the test.
- Swap in a stronger primality test if the modulus range grows much larger.

## Common Pitfalls

- Checking every exponent from `1` to `p - 2` instead of only the factor-derived exponents.
- Forgetting the edge case `p = 2`, whose primitive root is `1`.
- Factoring the candidate instead of factoring `p - 1`, which tests the wrong quantity entirely.
