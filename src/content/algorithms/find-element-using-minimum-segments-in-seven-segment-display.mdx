## Why It Surfaces

Given digit segment costs (`0->6, 1->2, 2->5, 3->5, 4->4, 5->5, 6->6, 7->3, 8->7, 9->6`), compute each number's total display cost and pick the earliest index with the minimum total.

## Worked Example

- `[1, 2, 3, 4, 5]` -> `1` (uses 2 segments)
- `[489, 206, 745, 123, 756]` -> `745` (minimum segment sum among entries)

```js
function findElementUsingMinimumSegmentsInSevenSegmentDisplay(arr) {
  if (!Array.isArray(arr) || arr.length === 0) {
    throw new RangeError('Expected a non-empty array of natural numbers');
  }

  const seg = [6, 2, 5, 5, 4, 5, 6, 3, 7, 6];

  const segmentCost = (value) => {
    if (!Number.isInteger(value) || value < 0) {
      throw new RangeError('Array must contain non-negative integers only');
    }

    if (value === 0) {
      return seg[0];
    }

    let x = value;
    let total = 0;
    while (x > 0) {
      total += seg[x % 10];
      x = Math.floor(x / 10);
    }
    return total;
  };

  let bestIndex = 0;
  let bestCost = segmentCost(arr[0]);

  for (let i = 1; i < arr.length; i += 1) {
    const cost = segmentCost(arr[i]);
    if (cost < bestCost) {
      bestCost = cost;
      bestIndex = i;
    }
  }

  return arr[bestIndex];
}
```
