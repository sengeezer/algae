## Why It Surfaces

This is a classic local-accounting problem: every occupied cell contributes four edges, and each shared side with another occupied cell removes one exposed edge. Summing these contributions gives the total perimeter directly.

## Worked Example

For matrix `[[1, 0], [1, 1]]`, the perimeter is `8`.

```js
function findPerimeterOfShapesFormedWith1sInBinaryMatrix(matrix) {
  if (!Array.isArray(matrix) || matrix.length === 0 || !Array.isArray(matrix[0])) {
    throw new RangeError("Expected a non-empty 2D matrix");
  }

  const rows = matrix.length;
  const columns = matrix[0].length;
  let perimeter = 0;

  for (let row = 0; row < rows; row += 1) {
    if (!Array.isArray(matrix[row]) || matrix[row].length !== columns) {
      throw new RangeError("Matrix must be rectangular");
    }

    for (let column = 0; column < columns; column += 1) {
      if (matrix[row][column] !== 0 && matrix[row][column] !== 1) {
        throw new RangeError("Matrix values must be binary (0 or 1)");
      }

      if (matrix[row][column] === 1) {
        let neighbors = 0;
        if (row > 0 && matrix[row - 1][column] === 1) neighbors += 1;
        if (row + 1 < rows && matrix[row + 1][column] === 1) neighbors += 1;
        if (column > 0 && matrix[row][column - 1] === 1) neighbors += 1;
        if (column + 1 < columns && matrix[row][column + 1] === 1) neighbors += 1;
        perimeter += 4 - neighbors;
      }
    }
  }

  return perimeter;
}
```

## Follow-up Questions

## Common Pitfalls

- Counting diagonal neighbors, which do not share perimeter edges.
- Assuming rows are uniform without validating rectangular shape.
- Forgetting to validate binary input values.
