## Problem

Convert an infix expression into prefix notation.

## Approach

1. Reverse the infix string and swap `(` with `)`.
2. Convert this transformed expression to postfix.
3. Reverse postfix output to get prefix.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function precedence(op) {
  if (op === '^') return 3;
  if (op === '*' || op === '/') return 2;
  if (op === '+' || op === '-') return 1;
  return 0;
}

function infixToPostfix(expr) {
  const stack = [];
  let out = '';

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      out += ch;
    } else if (ch === '(') {
      stack.push(ch);
    } else if (ch === ')') {
      while (stack.length && stack[stack.length - 1] !== '(') out += stack.pop();
      stack.pop();
    } else {
      while (stack.length && precedence(stack[stack.length - 1]) >= precedence(ch)) out += stack.pop();
      stack.push(ch);
    }
  }

  while (stack.length) out += stack.pop();
  return out;
}

function infixToPrefix(expr) {
  const reversed = expr
    .split('')
    .reverse()
    .map((ch) => (ch === '(' ? ')' : ch === ')' ? '(' : ch))
    .join('');

  const postfix = infixToPostfix(reversed);
  return postfix.split('').reverse().join('');
}
```
