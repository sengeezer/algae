## Problem

Given a linked list and an integer k, reverse every k nodes in the linked list and return the modified list. If the number of nodes is not a multiple of k, the remaining nodes should remain in their original order.

## Approach

1. Divide the linked list into groups of k nodes.
2. Reverse each group in place using pointer manipulation.
3. Connect the reversed groups by adjusting tail pointers.
4. Handle the final partial group (if any) by leaving it unchanged.

**Time Complexity**: O(n) for single pass with reversals
**Space Complexity**: O(1) auxiliary space

## Implementation

```javascript
class ListNode {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function reverseKGroup(head, k) {
  // Check if k nodes are available
  let current = head;
  for (let i = 0; i < k && current !== null; i++) {
    current = current.next;
  }

  // If fewer than k nodes remain, return head as-is
  if (current === null && head !== null) {
    // Count remaining nodes
    let count = 0;
    let temp = head;
    while (temp !== null) {
      count++;
      temp = temp.next;
    }
    if (count < k) {
      return head;
    }
  }

  // Reverse first k nodes
  let prev = null;
  current = head;
  for (let i = 0; i < k && current !== null; i++) {
    const next = current.next;
    current.next = prev;
    prev = current;
    current = next;
  }

  // Recursively reverse remaining groups
  head.next = reverseKGroup(current, k);
  return prev;
}
```

```typescript
class ListNode {
  data: number;

  constructor(data: number) {

  }
}

function reverseKGroup(head: ListNode | null, k: number): ListNode | null {
  // Check if k nodes are available
  let current = head;
  for (let i = 0; i < k && current !== null; i++) {
    current = current.next;
  }

  // If fewer than k nodes remain, return head as-is
  if (current === null && head !== null) {
    let count = 0;
    let temp = head;
    while (temp !== null) {
      count++;
      temp = temp.next;
    }
    if (count < k) {
      return head;
    }
  }

  // Reverse first k nodes
  let prev: ListNode | null = null;
  current = head;
  for (let i = 0; i < k && current !== null; i++) {
    const next = current.next;
    current.next = prev;
    prev = current;
    current = next;
  }

  // Recursively reverse remaining groups
  if (head !== null) {
    head.next = reverseKGroup(current, k);
  }
  return prev;
}
```
