## Problem

Convert a valid prefix expression into infix expression.

## Approach

Scan right to left. Push operands. For each operator, pop two operands `a` and `b`, form `(a op b)`, and push back.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function isOperator(ch) {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function prefixToInfix(expr) {
  const stack = [];

  for (let i = expr.length - 1; i >= 0; i--) {
    const ch = expr[i];

    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const a = stack.pop();
      const b = stack.pop();
      stack.push(`(${a}${ch}${b})`);
    }
  }

  return stack.pop() || '';
}
```

function isOperator(ch: string): boolean {
  return ['+', '-', '*', '/', '^'].includes(ch);

function prefixToInfix(expr: string): string {
  const stack: string[] = [];

  for (let i = expr.length - 1; i >= 0; i--) {
    const ch = expr[i];

    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const a = stack.pop()!;
      const b = stack.pop()!;
      stack.push(`(${a}${ch}${b})`);
    }
  }

  return stack.pop() ?? '';
}
