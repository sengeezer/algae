## Problem

For every array element, find the nearest smaller element on the left and the nearest smaller element on the right, then return the maximum absolute difference across all positions.

## Approach

Use increasing stacks to compute nearest smaller values on both sides, then scan once for the maximum absolute difference.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function maxDiffNearestSmaller(arr) {
  const n = arr.length;
  const left = new Array(n).fill(0);
  const right = new Array(n).fill(0);
  const stack = [];

  for (let i = 0; i < n; i++) {
    while (stack.length && stack[stack.length - 1] >= arr[i]) stack.pop();
    left[i] = stack.length ? stack[stack.length - 1] : 0;
    stack.push(arr[i]);
  }

  stack.length = 0;

  for (let i = n - 1; i >= 0; i--) {
    while (stack.length && stack[stack.length - 1] >= arr[i]) stack.pop();
    right[i] = stack.length ? stack[stack.length - 1] : 0;
    stack.push(arr[i]);
  }

  let best = 0;
  for (let i = 0; i < n; i++) {
    best = Math.max(best, Math.abs(left[i] - right[i]));
  }
  return best;
}
```

```typescript
  }

  let best = 0;

    best = Math.max(best, Math.abs(left[i] - right[i]));
  }
  return best;
}
```
