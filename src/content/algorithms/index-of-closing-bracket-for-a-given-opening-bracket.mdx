## Problem

Given an expression and an index of an opening parenthesis `'('`, return the index of its matching closing parenthesis `')'`.

## Approach

Scan expression with a stack of opening indices. On closing bracket, pair with stack top and store mapping.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function buildClosingIndexMap(expr) {
  const stack = [];
  const closeOfOpen = new Map();

  for (let i = 0; i < expr.length; i++) {
    if (expr[i] === '(') {
      stack.push(i);
    } else if (expr[i] === ')') {
      const open = stack.pop();
      closeOfOpen.set(open, i);
    }
  }

  return closeOfOpen;
}

function closingIndexForOpen(expr, openIndex) {
  if (openIndex < 0 || openIndex >= expr.length || expr[openIndex] !== '(') {
    return -1;
  }

  const map = buildClosingIndexMap(expr);
  return map.has(openIndex) ? map.get(openIndex) : -1;
}
```

```typescript
function buildClosingIndexMap(expr: string): Map<number, number> {
  const stack: number[] = [];
  const closeOfOpen = new Map<number, number>();

  for (let i = 0; i < expr.length; i++) {
    if (expr[i] === '(') {
      stack.push(i);
    } else if (expr[i] === ')') {
      const open = stack.pop()!;
      closeOfOpen.set(open, i);
    }
  }

  return closeOfOpen;
}

function closingIndexForOpen(expr: string, openIndex: number): number {
  if (openIndex < 0 || openIndex >= expr.length || expr[openIndex] !== '(') {
    return -1;
  }

  const map = buildClosingIndexMap(expr);
  return map.get(openIndex) ?? -1;
}
```
