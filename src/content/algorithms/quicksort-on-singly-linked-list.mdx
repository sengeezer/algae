## Problem

Sort a singly linked list using the QuickSort idea of partitioning around a pivot and recursively sorting smaller and larger partitions.

## Approach

Use the head as the pivot. Traverse the remaining nodes and split them into `smaller` and `greaterOrEqual` lists. Recursively sort both lists, then concatenate:

`sortedSmaller -> pivot -> sortedGreaterOrEqual`

This preserves the QuickSort partition idea while respecting linked-list pointer traversal.

**Time Complexity**: O(n log n) average, O(n^2) worst case
**Space Complexity**: O(log n) average recursion stack

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function concatenate(leftHead, pivot, rightHead) {
  pivot.next = rightHead;

  if (leftHead === null) {
    return pivot;
  }

  let tail = leftHead;
  while (tail.next !== null) {
    tail = tail.next;
  }

  tail.next = pivot;
  return leftHead;
}

function quickSortOnSinglyLinkedList(head) {
  if (head === null || head.next === null) {
    return head;
  }

  const pivot = head;
  let current = head.next;
  pivot.next = null;

  let smallerHead = null;
  let greaterHead = null;

  while (current !== null) {
    const next = current.next;

    if (current.data < pivot.data) {
      current.next = smallerHead;
      smallerHead = current;
    } else {
      current.next = greaterHead;
      greaterHead = current;
    }

    current = next;
  }

  const sortedSmaller = quickSortOnSinglyLinkedList(smallerHead);
  const sortedGreater = quickSortOnSinglyLinkedList(greaterHead);

  return concatenate(sortedSmaller, pivot, sortedGreater);
}
```
