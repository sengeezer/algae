## Why It Surfaces

For 3D vectors `A = [a1, a2, a3]` and `B = [b1, b2, b3]`:

- Dot product: `a1*b1 + a2*b2 + a3*b3`
- Cross product:
  - `x = a2*b3 - a3*b2`
  - `y = a3*b1 - a1*b3`
  - `z = a1*b2 - a2*b1`

## Worked Example

- `A = [3, -5, 4]`
- `B = [2, 6, 5]`
- Dot product: `-4`
- Cross product: `[-49, -7, 28]`

```js
function programForDotProductAndCrossProductOfTwoVectors(vectorA, vectorB) {
  if (!Array.isArray(vectorA) || !Array.isArray(vectorB) || vectorA.length !== 3 || vectorB.length !== 3) {
    throw new RangeError('Expected two 3D vectors');
  }

  const dotProduct =
    vectorA[0] * vectorB[0] +
    vectorA[1] * vectorB[1] +
    vectorA[2] * vectorB[2];

  const crossProduct = [
    vectorA[1] * vectorB[2] - vectorA[2] * vectorB[1],
    vectorA[2] * vectorB[0] - vectorA[0] * vectorB[2],
    vectorA[0] * vectorB[1] - vectorA[1] * vectorB[0],
  ];

  return { dotProduct, crossProduct };
}
```

export function programForDotProductAndCrossProductOfTwoVectors(
  vectorA: number[],
  vectorB: number[],
): { dotProduct: number; crossProduct: [number, number, number] } {
  if (!Array.isArray(vectorA) || !Array.isArray(vectorB) || vectorA.length !== 3 || vectorB.length !== 3) {
    throw new RangeError('Expected two 3D vectors');
  }

  const dotProduct =
    vectorA[0] * vectorB[0] +
    vectorA[1] * vectorB[1] +
    vectorA[2] * vectorB[2];

  const crossProduct: [number, number, number] = [
    vectorA[1] * vectorB[2] - vectorA[2] * vectorB[1],
    vectorA[2] * vectorB[0] - vectorA[0] * vectorB[2],
    vectorA[0] * vectorB[1] - vectorA[1] * vectorB[0],
  ];

  return { dotProduct, crossProduct };
}

## Common Pitfalls

- Swapping the second and third cross-product terms.
- Using vectors of the wrong dimension.
- Forgetting that cross product is anti-commutative.
