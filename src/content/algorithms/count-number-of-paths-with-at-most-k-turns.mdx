## Why It Surfaces

This is a useful follow-up to plain path counting because the usual two-dimensional recurrence is no longer enough. Whether you can spend the next move depends on the direction you used to get here, so the DP state must remember the last move as well as the remaining turn budget.

## Worked Example

In a `3 x 3` grid with at most `2` turns, the valid paths are `RRDD`, `RDDR`, `DRRD`, and `DDRR`, so the answer is `4`. The alternating paths `RDRD` and `DRDR` each use `3` turns and are excluded.

function validateGridArguments(rows, cols, maxTurns) {
  if (!Number.isInteger(rows) || rows <= 0) {
    throw new TypeError('rows must be a positive integer');
  }

  if (!Number.isInteger(cols) || cols <= 0) {
    throw new TypeError('cols must be a positive integer');
  }

  if (!Number.isInteger(maxTurns) || maxTurns < 0) {
    throw new TypeError('maxTurns must be a non-negative integer');
  }
}

function countNumberOfPathsWithAtMostKTurns(rows, cols, maxTurns) {
  validateGridArguments(rows, cols, maxTurns);

  const START = 'start';
  const RIGHT = 'right';
  const DOWN = 'down';
  const memo = new Map();

  function search(row, col, remainingTurns, direction) {
    if (row === rows - 1 && col === cols - 1) {
      return 1;
    }

    const key = `${row},${col},${remainingTurns},${direction}`;
    if (memo.has(key)) {
      return memo.get(key);
    }

    let total = 0;

    if (col + 1 < cols) {
      const nextTurns = direction === DOWN ? remainingTurns - 1 : remainingTurns;
      if (nextTurns >= 0) {
        total += search(row, col + 1, nextTurns, RIGHT);
      }
    }

    if (row + 1 < rows) {
      const nextTurns = direction === RIGHT ? remainingTurns - 1 : remainingTurns;
      if (nextTurns >= 0) {
        total += search(row + 1, col, nextTurns, DOWN);
      }
    }

    memo.set(key, total);
    return total;
  }

  return search(0, 0, maxTurns, START);
}

function validateGridArguments(rows: number, cols: number, maxTurns: number): void {
  if (!Number.isInteger(rows) || rows <= 0) {
    throw new TypeError('rows must be a positive integer');
  }

  if (!Number.isInteger(cols) || cols <= 0) {
    throw new TypeError('cols must be a positive integer');
  }

  if (!Number.isInteger(maxTurns) || maxTurns < 0) {
    throw new TypeError('maxTurns must be a non-negative integer');
  }
}

export function countNumberOfPathsWithAtMostKTurns(
  rows: number,
  cols: number,
  maxTurns: number,
): number {
  validateGridArguments(rows, cols, maxTurns);

  const START = 'start';
  const RIGHT = 'right';
  const DOWN = 'down';
  const memo = new Map<string, number>();

  const search = (
    row: number,
    col: number,
    remainingTurns: number,
    direction: typeof START | typeof RIGHT | typeof DOWN,
  ): number => {
    if (row === rows - 1 && col === cols - 1) {
      return 1;
    }

    const key = `${row},${col},${remainingTurns},${direction}`;
    const cached = memo.get(key);
    if (cached !== undefined) {
      return cached;
    }

    let total = 0;

    if (col + 1 < cols) {
      const nextTurns = direction === DOWN ? remainingTurns - 1 : remainingTurns;
      if (nextTurns >= 0) {
        total += search(row, col + 1, nextTurns, RIGHT);
      }
    }

    if (row + 1 < rows) {
      const nextTurns = direction === RIGHT ? remainingTurns - 1 : remainingTurns;
      if (nextTurns >= 0) {
        total += search(row + 1, col, nextTurns, DOWN);
      }
    }

    memo.set(key, total);
    return total;
  };

  return search(0, 0, maxTurns, START);
}

## Follow-up Questions

- How would you change the state if the path needed exactly k turns instead of at most k?
- Why does ordinary Unique Paths fail to capture this constraint without a direction dimension?
- How would you extend the model to allow diagonal moves or blocked cells?

## Common Pitfalls

- Counting turns from the start cell before any direction has been chosen.
- Using only row and column as state even though the next move cost depends on the last direction.
- Forgetting that single-row and single-column grids can have valid zero-turn paths.
