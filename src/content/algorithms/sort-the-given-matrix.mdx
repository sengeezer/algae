## Why It Surfaces

This question tests whether you interpret the sorting requirement globally instead of row by row. Once the prompt defines strict order across row boundaries, the simplest correct strategy is to flatten, sort once, and rebuild the matrix in row-major order.

## Worked Example

For `[[5, 4, 7], [1, 3, 8], [2, 9, 6]]`, sorting all values produces `[1, 2, 3, 4, 5, 6, 7, 8, 9]`. Writing them back row by row yields `[[1, 2, 3], [4, 5, 6], [7, 8, 9]]`.

function validateNumericMatrix(matrix) {
  if (!Array.isArray(matrix) || matrix.length === 0 || !matrix.every((row) => Array.isArray(row))) {
    throw new TypeError('matrix must be a non-empty 2D array');
  }

  const width = matrix[0].length;
  if (width === 0) {
    throw new TypeError('matrix rows must not be empty');
  }

  if (!matrix.every((row) => row.length === width && row.every((value) => Number.isFinite(value)))) {
    throw new TypeError('matrix rows must have equal length and contain only finite numbers');
  }
}

function sortTheGivenMatrix(matrix) {
  validateNumericMatrix(matrix);

  const rows = matrix.length;
  const cols = matrix[0].length;
  const sortedValues = matrix.flat().sort((left, right) => left - right);

  return Array.from({ length: rows }, (_, rowIndex) =>
    Array.from({ length: cols }, (_, colIndex) => sortedValues[rowIndex * cols + colIndex]),
  );
}

  [1, 3, 8],

]);
console.assert(

    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9],
  ]),
  '3x3 matrix should rebuild into strict row-major order',
);

const sortedRectangle = sortTheGivenMatrix([

]);
console.assert(
  JSON.stringify(sortedRectangle) === JSON.stringify([

## Common Pitfalls

- Sorting rows one at a time and ignoring ordering across row boundaries.
- Reconstructing with the wrong row-major index formula.
- Mutating the input matrix when a pure transformed result is easier to verify.
