## Why It Surfaces

Spiral traversal is the natural follow-up to perimeter traversal. It tests whether you can manage four moving boundaries carefully enough to emit each ring exactly once and stop cleanly when the boundaries cross.

## Worked Example

For `[[1, 2, 3], [4, 5, 6], [7, 8, 9]]`, the first ring gives `[1, 2, 3, 6, 9, 8, 7, 4]`. After shrinking the boundaries, only the center `5` remains, so the full spiral order is `[1, 2, 3, 6, 9, 8, 7, 4, 5]`.

function validateNumericMatrix(matrix) {
  if (!Array.isArray(matrix) || matrix.length === 0 || !matrix.every((row) => Array.isArray(row))) {
    throw new TypeError('matrix must be a non-empty 2D array');
  }

  const width = matrix[0].length;
  if (width === 0) {
    throw new TypeError('matrix rows must not be empty');
  }

  if (!matrix.every((row) => row.length === width && row.every((value) => Number.isFinite(value)))) {
    throw new TypeError('matrix rows must have equal length and contain only finite numbers');
  }
}

function printAGivenMatrixInSpiralForm(matrix) {
  validateNumericMatrix(matrix);

  const result = [];
  let top = 0;
  let bottom = matrix.length - 1;
  let left = 0;
  let right = matrix[0].length - 1;

  while (top <= bottom && left <= right) {
    for (let col = left; col <= right; col += 1) {
      result.push(matrix[top][col]);
    }
    top += 1;

    for (let row = top; row <= bottom; row += 1) {
      result.push(matrix[row][right]);
    }
    right -= 1;

    if (top <= bottom) {
      for (let col = right; col >= left; col -= 1) {
        result.push(matrix[bottom][col]);
      }
      bottom -= 1;
    }

    if (left <= right) {
      for (let row = bottom; row >= top; row -= 1) {
        result.push(matrix[row][left]);
      }
      left += 1;
    }
  }

  return result;
}

## Common Pitfalls

- Forgetting the boundary-crossing checks before traversing the bottom row or left column.
- Repeating cells when the remaining submatrix collapses to a single row or column.
- Confusing spiral traversal with a one-step ring rotation.
