## Why It Surfaces

This problem is really about understanding what the hull depends on. Removing an interior point changes nothing, but removing a hull vertex can expose a new boundary and force a full rebuild.

## Worked Example

For the initial set `(-2, 8)`, `(-1, 2)`, `(0, 1)`, `(1, 0)`, `(-3, 0)`, `(-1, -9)`, `(2, -6)`, `(3, 0)`, `(5, 3)`, `(2, 5)`, the hull is `(-3, 0)`, `(-1, -9)`, `(2, -6)`, `(5, 3)`, `(-2, 8)`. Removing `(-2, 8)` changes the hull to `(-3, 0)`, `(-1, -9)`, `(2, -6)`, `(5, 3)`, `(2, 5)`.

function comparePoints(left, right) {
  if (left.x !== right.x) {
    return left.x - right.x;
  }
  return left.y - right.y;
}

function cross(origin, first, second) {
  return (first.x - origin.x) * (second.y - origin.y) - (first.y - origin.y) * (second.x - origin.x);
}

function buildConvexHull(points) {
  const sorted = [...new Map(points.map((point) => [`${point.x},${point.y}`, point])).values()].sort(comparePoints);
  if (sorted.length <= 1) {
    return sorted;
  }

  const lower = [];
  for (const point of sorted) {
    while (lower.length >= 2 && cross(lower[lower.length - 2], lower[lower.length - 1], point) <= 0) {
      lower.pop();
    }
    lower.push(point);
  }

  const upper = [];
  for (let index = sorted.length - 1; index >= 0; index -= 1) {
    const point = sorted[index];
    while (upper.length >= 2 && cross(upper[upper.length - 2], upper[upper.length - 1], point) <= 0) {
      upper.pop();
    }
    upper.push(point);
  }

  lower.pop();
  upper.pop();
  return [...lower, ...upper];
}

function deletingPointsFromConvexHull(points, pointToRemove) {
  if (!Array.isArray(points) || points.length < 3) {
    throw new RangeError("Convex hull requires at least three points");
  }

  const initialHull = buildConvexHull(points);
  const removedFromHull = initialHull.some((point) => point.x === pointToRemove.x && point.y === pointToRemove.y);

  const remainingPoints = [];
  let removed = false;
  for (const point of points) {
    if (!removed && point.x === pointToRemove.x && point.y === pointToRemove.y) {
      removed = true;
      continue;
    }
    remainingPoints.push(point);
  }

  const updatedHull = removedFromHull ? buildConvexHull(remainingPoints) : initialHull;
  return { initialHull, updatedHull, removedFromHull };
}

## Follow-up Questions

- Why is it safe to leave the hull unchanged when the removed point was interior?
- How would you support many deletion queries without rebuilding from scratch each time?

## Common Pitfalls

- Recomputing the hull unconditionally even when the deleted point was never on the boundary.
- Forgetting to remove only one matching point when duplicates exist.
- Returning the updated hull in an unstable order that makes comparisons hard.
