## Overview

A permutation is stack-sortable if pushing elements through one stack (with optional pops) can output `1..n` in sorted order.

The number of stack-sortable permutations of size `n` is the nth Catalan number.

## Approach

```js
function catalan(n) {
  if (!Number.isInteger(n) || n < 0) return -1n;
  let result = 1n;
  for (let i = 1n; i <= BigInt(n); i++) {
    result = (result * (4n * i - 2n)) / (i + 1n);
  }
  return result;
}

function countStackSortablePermutations(n) {
  return catalan(n);
}

function isStackSortable(perm) {
  const stack = [];
  let need = 1;

  for (const x of perm) {
    stack.push(x);
    while (stack.length > 0 && stack[stack.length - 1] === need) {
      stack.pop();
      need += 1;
    }
  }

  return stack.length === 0;
}

console.assert(countStackSortablePermutations(0) === 1n, "n=0 => 1");
console.assert(countStackSortablePermutations(1) === 1n, "n=1 => 1");
console.assert(countStackSortablePermutations(3) === 5n, "n=3 => 5");
console.assert(countStackSortablePermutations(4) === 14n, "n=4 => 14");
console.assert(isStackSortable([2, 1, 3]) === true, "[2,1,3] is stack-sortable");
console.assert(isStackSortable([2, 3, 1]) === false, "[2,3,1] is not stack-sortable");
```

## Notes

- A classic characterization: stack-sortable permutations avoid the pattern `231`.
- Counting them yields the same Catalan sequence seen in Dyck structures.
