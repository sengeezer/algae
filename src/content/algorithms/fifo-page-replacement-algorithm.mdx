## Problem

Given a page reference string and a frame count, return the number of page faults under FIFO replacement.

## Approach

Track pages currently in memory with a set and their arrival order with a queue. On a miss, evict the oldest page when frames are full.

**Time Complexity**: O(n)
**Space Complexity**: O(frames)

## Implementation

```javascript
function fifoPageReplacement(pages, frameCount) {
  const frames = new Set();
  const order = [];
  let faults = 0;

  for (const page of pages) {
    if (frames.has(page)) continue;

    faults += 1;

    if (frames.size === frameCount) {
      const oldest = order.shift();
      frames.delete(oldest);
    }

    frames.add(page);
    order.push(page);
  }

  return faults;
}
```

```typescript
function fifoPageReplacement(pages: number[], frameCount: number): number {
  const frames = new Set<number>();
  const order: number[] = [];
  let faults = 0;

  for (const page of pages) {
    if (frames.has(page)) continue;

    faults += 1;

    if (frames.size === frameCount) {
      const oldest = order.shift()!;
      frames.delete(oldest);
    }

    frames.add(page);
    order.push(page);
  }

  return faults;
}
```
