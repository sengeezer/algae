## Problem

Print the optimal sequence of moves to solve Tower of Hanoi iteratively for `n` disks.

## Approach

Use three rods as stacks and perform `2^n - 1` moves. At each step, choose move between specific rod pairs in cyclic order.

**Time Complexity**: O(2^n)
**Space Complexity**: O(n)

## Implementation

```javascript
function moveDisk(from, to, fromName, toName, moves) {
  const fromTop = from.length ? from[from.length - 1] : Infinity;
  const toTop = to.length ? to[to.length - 1] : Infinity;

  if (fromTop < toTop) {
    const disk = from.pop();
    to.push(disk);
    moves.push(`Move disk ${disk} from ${fromName} to ${toName}`);
  } else {
    const disk = to.pop();
    from.push(disk);
    moves.push(`Move disk ${disk} from ${toName} to ${fromName}`);
  }
}

function iterativeTowerOfHanoi(n) {
  const src = [];
  const aux = [];
  const dst = [];
  const moves = [];

  for (let d = n; d >= 1; d--) src.push(d);

  let sName = 'A';
  let aName = 'B';
  let dName = 'C';

  if (n % 2 === 0) {
    [aName, dName] = [dName, aName];
  }

  const totalMoves = Math.pow(2, n) - 1;

  for (let i = 1; i <= totalMoves; i++) {
    if (i % 3 === 1) moveDisk(src, dst, sName, dName, moves);
    else if (i % 3 === 2) moveDisk(src, aux, sName, aName, moves);
    else moveDisk(aux, dst, aName, dName, moves);
  }

  return moves;
}
```
