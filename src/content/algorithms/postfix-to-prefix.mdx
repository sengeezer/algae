## Problem

Convert a valid postfix expression into prefix expression.

## Approach

Scan left to right. Push operands. On operator, pop `right` then `left`, and push `operator + left + right`.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function isOperator(ch) {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function postfixToPrefix(expr) {
  const stack = [];

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const right = stack.pop();
      const left = stack.pop();
      stack.push(ch + left + right);
    }
  }

  return stack.pop() || '';
}
```

```typescript
function isOperator(ch: string): boolean {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function postfixToPrefix(expr: string): string {
  const stack: string[] = [];

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const right = stack.pop()!;
      const left = stack.pop()!;
      stack.push(ch + left + right);
    }
  }

  return stack.pop() ?? '';
}
```
