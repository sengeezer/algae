## Why It Surfaces

Triangular numbers appear in array problems, combinatorial counting, and figurate-number classifications. The sequence is fundamental to understanding polygonal numbers and their relationships.

## Worked Example

The first five triangular numbers are: 1, 3 (1+2), 6 (1+2+3), 10 (1+2+3+4), and 15 (1+2+3+4+5). The 4th triangular number is 10.

function triangularNumber(n) {
  if (!Number.isInteger(n) || n < 1) {
    throw new RangeError('n must be a positive integer');
  }

  return (n * (n + 1)) / 2;
}

function generateTriangularNumbers(limit) {
  if (!Number.isInteger(limit) || limit < 1) {
    throw new RangeError('Limit must be a positive integer');
  }

  const triangular = [];
  let n = 1;

  while (true) {
    const tri = triangularNumber(n);
    if (tri > limit) {
      break;
    }
    triangular.push(tri);
    n += 1;
  }

  return triangular;
}

function isTriangularNumber(num) {
  if (!Number.isInteger(num) || num < 1) {
    return false;
  }

  const discriminant = 1 + 8 * num;
  const sqrtDiscriminant = Math.sqrt(discriminant);

  if (!Number.isInteger(sqrtDiscriminant)) {
    return false;
  }

  const n = (-1 + sqrtDiscriminant) / 2;
  return Number.isInteger(n) && n > 0;
}

console.assert(triangularNumber(5) === 15, 'T(5) = 15');

    JSON.stringify([1, 3, 6, 10, 15, 21, 28]),

console.assert(isTriangularNumber(10) === true, '10 is triangular');
console.assert(isTriangularNumber(11) === false, '11 is not triangular');
console.assert(isTriangularNumber(21) === true, '21 is triangular');

## Key Insights

- The formula T(n) = n times (n+1) divided by 2 provides constant-time lookup.
- Triangular numbers are cumulative sums of the first n natural numbers.
- Tetrahedral numbers are built by summing consecutive triangular numbers.
- Testing membership requires solving a quadratic equation and checking if the root is positive.
