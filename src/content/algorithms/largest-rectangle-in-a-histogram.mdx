## Problem

Given bar heights of a histogram, return the area of the largest rectangle that can be formed.

## Approach

Maintain an increasing stack of indices. When a lower bar appears, pop taller bars and compute the rectangle where each popped bar is the limiting height.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function largestRectangleArea(heights) {
  const extended = [...heights, 0];
  const stack = [];
  let best = 0;

  for (let i = 0; i < extended.length; i++) {
    while (stack.length && extended[stack[stack.length - 1]] > extended[i]) {
      const height = extended[stack.pop()];
      const left = stack.length ? stack[stack.length - 1] : -1;
      const width = i - left - 1;
      best = Math.max(best, height * width);
    }
    stack.push(i);
  }

  return best;
}
```

```typescript
function largestRectangleArea(heights: number[]): number {
  const extended = [...heights, 0];
  const stack: number[] = [];
  let best = 0;

  for (let i = 0; i < extended.length; i++) {
    while (stack.length > 0 && extended[stack[stack.length - 1]] > extended[i]) {
      const height = extended[stack.pop()!];
      const left = stack.length > 0 ? stack[stack.length - 1] : -1;
      const width = i - left - 1;
      best = Math.max(best, height * width);
    }
    stack.push(i);
  }

  return best;
}
```
