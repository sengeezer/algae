## Problem

Reverse a singly linked list so that the last node becomes the head and pointers all point in the opposite direction.

## Approach

**Iterative Pointer Reversal**

Maintain three pointers: `prev` (initially null), `current` (initially head), and `next` (to save the next node before overwriting). Traverse the list once, at each node reverse the pointer by setting `current.next = prev`, then advance `prev` and `current` one step. At the end, `prev` points to the new head.

**Time Complexity**: O(n)
**Space Complexity**: O(1)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

function validateLinkedList(head) {
  return head === null || (head instanceof Node);
}

function reverseLinkedList(head) {
  console.assert(validateLinkedList(head), 'Input must be a valid linked list head or null');

  let prev = null;
  let current = head;

  while (current !== null) {
    // Save next node before modifying current.next
    const next = current.next;

    // Reverse the link
    current.next = prev;

    // Move prev and current one step forward
    prev = current;
    current = next;
  }

  return prev; // New head
}
```

```typescript
class Node {
  data: number;
  next: Node | null;

  constructor(data: number) {
    this.data = data;
    this.next = null;
  }
}

function validateLinkedList(head: unknown): boolean {
  return head === null || head instanceof Node;
}

function reverseLinkedList(head: Node | null): Node | null {
  console.assert(validateLinkedList(head), 'Input must be a valid linked list head or null');

  let prev: Node | null = null;
  let current: Node | null = head;

  while (current !== null) {
    // Save next node before modifying current.next
    const next = current.next;

    // Reverse the link
    current.next = prev;

    // Move prev and current one step forward
    prev = current;
    current = next;
  }

  return prev; // New head
}
```

## Example

```javascript
// Build list: 1 -> 2 -> 3 -> 4 -> null
const head = new Node(1);
head.next = new Node(2);
head.next.next = new Node(3);
head.next.next.next = new Node(4);

const reversed = reverseLinkedList(head);
// Result: 4 -> 3 -> 2 -> 1 -> null
```
