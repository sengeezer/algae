## Problem

Given a linked list where each node has `next` and `random` pointers, create a deep copy of the list.

## Approach

1. Insert copied node after each original node.
2. Set copied random pointers: `copy.random = original.random.next`.
3. Detach interleaved structure into original and copied lists.

**Time Complexity**: O(n)
**Space Complexity**: O(1) auxiliary

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
    this.random = null;
  }
}

function cloneLinkedListWithRandomPointer(head) {
  if (head === null) return null;

  let current = head;
  while (current !== null) {
    const copy = new Node(current.data);
    copy.next = current.next;
    current.next = copy;
    current = copy.next;
  }

  current = head;
  while (current !== null) {
    if (current.random !== null) {
      current.next.random = current.random.next;
    }
    current = current.next.next;
  }

  const copyHead = head.next;
  current = head;

  while (current !== null) {
    const copy = current.next;
    current.next = copy.next;
    copy.next = copy.next !== null ? copy.next.next : null;
    current = current.next;
  }

  return copyHead;
}
```

```typescript
class Node {
  data: number;
  next: Node | null;
  random: Node | null;

  constructor(data: number) {
    this.data = data;
    this.next = null;
    this.random = null;
  }
}

function cloneLinkedListWithRandomPointer(head: Node | null): Node | null {
  if (head === null) return null;

  let current: Node | null = head;
  while (current !== null) {
    const copy = new Node(current.data);
    copy.next = current.next;
    current.next = copy;
    current = copy.next;
  }

  current = head;
  while (current !== null) {
    if (current.random !== null) {
      current.next!.random = current.random.next;
    }
    current = current.next!.next;
  }

  const copyHead = head.next;
  current = head;

  while (current !== null) {
    const copy = current.next!;
    current.next = copy.next;
    copy.next = copy.next !== null ? copy.next.next : null;
    current = current.next;
  }

  return copyHead;
}
```
