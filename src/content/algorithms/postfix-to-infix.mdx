## Problem

Convert a valid postfix expression to infix expression.

## Approach

Push operands. On each operator, pop `right` and `left`, create `(left op right)`, then push it back.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function isOperator(ch) {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function postfixToInfix(expr) {
  const stack = [];

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const right = stack.pop();
      const left = stack.pop();
      stack.push(`(${left}${ch}${right})`);
    }
  }

  return stack.pop() || '';
}
```

```typescript
function isOperator(ch: string): boolean {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function postfixToInfix(expr: string): string {
  const stack: string[] = [];

  for (const ch of expr) {
    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const right = stack.pop()!;
      const left = stack.pop()!;
      stack.push(`(${left}${ch}${right})`);
    }
  }

  return stack.pop() ?? '';
}
```
