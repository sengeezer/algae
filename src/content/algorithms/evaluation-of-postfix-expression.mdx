## Problem

Evaluate a valid postfix expression containing integer operands and operators `+`, `-`, `*`, `/`.

## Approach

Push operands to stack. On operator, pop `right` then `left`, compute `left op right`, and push result.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function evalPostfix(tokens) {
  const stack = [];

  for (const token of tokens) {
    if (/^-?\d+$/.test(token)) {
      stack.push(Number(token));
      continue;
    }

    const right = stack.pop();
    const left = stack.pop();

    if (token === '+') stack.push(left + right);
    else if (token === '-') stack.push(left - right);
    else if (token === '*') stack.push(left * right);
    else if (token === '/') stack.push(Math.trunc(left / right));
  }

  return stack.pop();
}
```

```typescript
function evalPostfix(tokens: string[]): number {
  const stack: number[] = [];

  for (const token of tokens) {
    if (/^-?\d+$/.test(token)) {
      stack.push(Number(token));
      continue;
    }

    const right = stack.pop()!;
    const left = stack.pop()!;

    if (token === '+') stack.push(left + right);
    else if (token === '-') stack.push(left - right);
    else if (token === '*') stack.push(left * right);
    else if (token === '/') stack.push(Math.trunc(left / right));
  }

  return stack.pop()!;
}
```
