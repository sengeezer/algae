## Why It Surfaces

This is a compact bridge between geometry and number theory. The segment step vector `(dx, dy)` can be reduced to primitive direction `(dx / g, dy / g)` where `g = gcd(|dx|, |dy|)`, and exactly `g - 1` interior lattice points lie between endpoints.

## Worked Example

- Between `(2, 2)` and `(5, 5)`: `dx = 3`, `dy = 3`, `gcd = 3`, so interior count is `2`.
- Between `(1, 9)` and `(8, 16)`: `dx = 7`, `dy = 7`, `gcd = 7`, so interior count is `6`.

function gcd(a, b) {
  let left = Math.abs(a);
  let right = Math.abs(b);
  while (right !== 0) {
    const next = left % right;
    left = right;
    right = next;
  }
  return left;
}

function numberOfIntegralPointsBetweenTwoPoints(first, second) {
  if (
    !first || !second
    || !Number.isInteger(first.x) || !Number.isInteger(first.y)
    || !Number.isInteger(second.x) || !Number.isInteger(second.y)
  ) {
    throw new RangeError("Points must use integer coordinates");
  }

  const dx = Math.abs(second.x - first.x);
  const dy = Math.abs(second.y - first.y);
  if (dx === 0 && dy === 0) {
    return 0;
  }

  return Math.max(0, gcd(dx, dy) - 1);
}

  let left = Math.abs(a);
  let right = Math.abs(b);
  while (right !== 0) {
    const next = left % right;
    left = right;
    right = next;
  }
  return left;
}

export function numberOfIntegralPointsBetweenTwoPoints(first: Point, second: Point): number {

    !Number.isInteger(first.x) || !Number.isInteger(first.y)

  const dy = Math.abs(second.y - first.y);

    return 0;

}

- Using floating-point gcd logic instead of integer arithmetic.
- Forgetting the degenerate same-point case.
