## Problem

Given a queue and integer `k`, reverse only the first `k` elements and keep the rest in original order.

## Approach

1. Dequeue first `k` elements into a stack.
2. Pop stack back into queue (reversed segment).
3. Rotate remaining `n - k` elements to back.

**Time Complexity**: O(n)
**Space Complexity**: O(k)

## Implementation

```javascript
function reverseFirstK(queue, k) {
  const n = queue.length;
  if (k <= 0 || k > n) return queue;

  const stack = [];

  for (let i = 0; i < k; i++) {
    stack.push(queue.shift());
  }

  while (stack.length) {
    queue.push(stack.pop());
  }

  for (let i = 0; i < n - k; i++) {
    queue.push(queue.shift());
  }

  return queue;
}
```

```typescript
function reverseFirstK(queue: number[], k: number): number[] {
  const n = queue.length;
  if (k <= 0 || k > n) return queue;

  const stack: number[] = [];

  for (let i = 0; i < k; i++) {
    stack.push(queue.shift()!);
  }

  while (stack.length > 0) {
    queue.push(stack.pop()!);
  }

  for (let i = 0; i < n - k; i++) {
    queue.push(queue.shift()!);
  }

  return queue;
}
```
