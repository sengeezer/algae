## Problem

Given a doubly linked list and integer `n`, rotate the list left by `n` nodes.

## Approach

Normalize `n` by list length. Traverse to the n-th node, split list after it, then append the original first segment at the end of the second segment while preserving both `next` and `prev` pointers.

**Time Complexity**: O(n)
**Space Complexity**: O(1)

## Implementation

```javascript
class Node {
  constructor(data) {
    this.data = data;
    this.prev = null;
    this.next = null;
  }
}

function rotateDoublyLinkedListByNNodes(head, n) {
  if (head === null || head.next === null || n === 0) {
    return head;
  }

  let length = 1;
  let tail = head;

  while (tail.next !== null) {
    tail = tail.next;
    length++;
  }

  n = n % length;
  if (n === 0) {
    return head;
  }

  let split = head;
  for (let i = 1; i < n; i++) {
    split = split.next;
  }

  const newHead = split.next;
  split.next = null;
  newHead.prev = null;

  tail.next = head;
  head.prev = tail;

  return newHead;
}
```

class Node {
  data: number;
  prev: Node | null;
  next: Node | null;

  constructor(data: number) {
    this.data = data;
    this.prev = null;
    this.next = null;
  }
}

function rotateDoublyLinkedListByNNodes(head: Node | null, n: number): Node | null {
  if (head === null || head.next === null || n === 0) {
    return head;
  }

  let length = 1;
  let tail: Node = head;

  while (tail.next !== null) {
    tail = tail.next;
    length++;
  }

  n = n % length;
  if (n === 0) {
    return head;
  }

  let split: Node = head;
  for (let i = 1; i < n; i++) {
    split = split.next!;
  }

  const newHead = split.next!;
  split.next = null;
  newHead.prev = null;

  tail.next = head;
  head.prev = tail;

  return newHead;
}
