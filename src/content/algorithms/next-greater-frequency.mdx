## Problem

For each element, return the next element to its right that has a greater frequency in the array; otherwise `-1`.

## Approach

Build frequency map. Traverse right to left with a stack storing values that remain candidates by descending frequency.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function nextGreaterFrequency(arr) {
  const freq = new Map();
  for (const x of arr) freq.set(x, (freq.get(x) || 0) + 1);

  const stack = [];
  const ans = new Array(arr.length).fill(-1);

  for (let i = arr.length - 1; i >= 0; i--) {
    while (stack.length > 0 && freq.get(stack[stack.length - 1]) <= freq.get(arr[i])) {
      stack.pop();
    }

    ans[i] = stack.length === 0 ? -1 : stack[stack.length - 1];
    stack.push(arr[i]);
  }

  return ans;
}
```

  for (let i = arr.length - 1; i >= 0; i--) {
    while (stack.length > 0 && (freq.get(stack[stack.length - 1]) ?? 0) <= (freq.get(arr[i]) ?? 0)) {
      stack.pop();
    }

    ans[i] = stack.length === 0 ? -1 : stack[stack.length - 1];
    stack.push(arr[i]);
  }

  return ans;
}
