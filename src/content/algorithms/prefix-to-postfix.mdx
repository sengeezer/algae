## Problem

Convert a valid prefix expression into postfix expression.

## Approach

Scan from right to left:
- push operands,
- on operator, pop `a` and `b`, push `a + b + operator`.

**Time Complexity**: O(n)
**Space Complexity**: O(n)

## Implementation

```javascript
function isOperator(ch) {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function prefixToPostfix(expr) {
  const stack = [];

  for (let i = expr.length - 1; i >= 0; i--) {
    const ch = expr[i];

    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const a = stack.pop();
      const b = stack.pop();
      stack.push(a + b + ch);
    }
  }

  return stack.pop() || '';
}
```

```typescript
function isOperator(ch: string): boolean {
  return ['+', '-', '*', '/', '^'].includes(ch);
}

function prefixToPostfix(expr: string): string {
  const stack: string[] = [];

  for (let i = expr.length - 1; i >= 0; i--) {
    const ch = expr[i];

    if (/[a-zA-Z0-9]/.test(ch)) {
      stack.push(ch);
    } else if (isOperator(ch)) {
      const a = stack.pop()!;
      const b = stack.pop()!;
      stack.push(a + b + ch);
    }
  }

  return stack.pop() ?? '';
}
```
